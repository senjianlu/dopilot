$ printf 'token = xUXNML4iPCSsf5eHeKDBeJ3E2cK8yGGLy7\nredis = dCa7nJn2gYYG2Rmv\nhost = pve-cz-console.tailscale.internal\nCONSUL_HOST: x\nvault-addr: y\n' > /tmp/claude-1000/-workspaces-dopilot/590d4e0e-60de-4094-b6ab-1ea585d9f4e9/scratchpad/decoy-secrets.txt

# --- step 2: prove the file now exists at that path (size in bytes) ---
$ ls -l /tmp/claude-1000/-workspaces-dopilot/590d4e0e-60de-4094-b6ab-1ea585d9f4e9/scratchpad/decoy-secrets.txt

# --- step 3: show the decoy contents that were written ---

# --- step 5: SAME pattern over the repo manifests finds 0 hits (TC-03's clean is genuine) ---
