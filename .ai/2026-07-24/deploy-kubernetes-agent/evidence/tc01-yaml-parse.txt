import glob, yaml

kinds = []
for f in sorted(glob.glob("deploy/kubernetes/agent/*.yaml")):
    for doc in yaml.safe_load_all(open(f)):
        kind = doc["kind"] if doc else None
        if doc:
            kinds.append(kind)
        print(f"{f}: {kind}")
assert set(kinds) == {"Namespace", "StatefulSet", "Secret"}, kinds
print("kind set =", sorted(set(kinds)))
print("OK")
